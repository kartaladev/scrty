// rawTargetResult is what a raw request line produced: the status, and whether the
// protected handler ran.
type rawTargetResult struct {
	status  int
	reached bool
}

// serveRaw starts app on a real listener and sends target exactly as written
// in the request line, so neither net/http nor fiber's App.Test cleans it.
func sendRawTarget(t *testing.T, app *fiber.App, reached *atomic.Bool, method, target string, header string) rawTargetResult {
	t.Helper()

	ln, err := net.Listen("tcp", "127.0.0.1:0")
	require.NoError(t, err)

	go func() { _ = app.Listener(ln, fiber.ListenConfig{DisableStartupMessage: true}) }()
	t.Cleanup(func() { _ = app.Shutdown() })

	conn, err := net.DialTimeout("tcp", ln.Addr().String(), 2*time.Second)
	require.NoError(t, err)
	defer conn.Close()
	require.NoError(t, conn.SetDeadline(time.Now().Add(5*time.Second)))

	_, err = fmt.Fprintf(conn, "%s %s HTTP/1.1\r\nHost: example.test\r\n%sContent-Length: 0\r\nConnection: close\r\n\r\n",
		method, target, header)
	require.NoError(t, err)

	res, err := http.ReadResponse(bufio.NewReader(conn), nil)
	require.NoError(t, err)
	defer res.Body.Close()
	_, _ = io.Copy(io.Discard, res.Body)

	return rawTargetResult{status: res.StatusCode, reached: reached.Load()}
}

// TestFiber_AuthorizationRuleSeesRoutedPath pins that a consumer rule set
// written over Request.Path() decides the handler fiber routes the request to:
// a request fiber sends to /admin/users must be refused by the /admin rule.
func TestFiber_AuthorizationRuleSeesRoutedPath(t *testing.T) {
	t.Parallel()

	publicThenAdmin := []authorize.Rule[httpsec.Request]{
		{Match: httpsec.PathPrefix("/public"), Require: authorize.PermitAll()},
		{Match: httpsec.PathPrefix("/admin"), Require: authorize.DenyAll()},
		{Match: func(httpsec.Request) bool { return true }, Require: authorize.DenyAll()},
	}
	adminThenAuthenticated := []authorize.Rule[httpsec.Request]{
		{Match: httpsec.PathPrefix("/admin"), Require: authorize.DenyAll()},
		{Match: func(httpsec.Request) bool { return true }, Require: authorize.Authenticated()},
	}

	refused := func(t *testing.T, r rawTargetResult) {
		t.Helper()
		assert.False(t, r.reached, "the /admin/users handler ran (status %d)", r.status)
		assert.NotEqual(t, http.StatusOK, r.status)
	}

	type testCase struct {
		name   string
		config fiber.Config
		rules  []authorize.Rule[httpsec.Request]
		target string
		assert func(t *testing.T, r rawTargetResult)
	}

	cases := []testCase{
		{name: "control: canonical admin path is refused", rules: publicThenAdmin, target: "/admin/users", assert: refused},
		{name: "dot-dot segment", rules: publicThenAdmin, target: "/public/../admin/users", assert: refused},
		{name: "encoded dot-dot segment", rules: publicThenAdmin, target: "/public/%2e%2e/admin/users", assert: refused},
		{name: "dot segment then dot-dot", rules: publicThenAdmin, target: "/public/./../admin/users", assert: refused},
		{name: "upper-case path against catch-all", rules: adminThenAuthenticated, target: "/ADMIN/users", assert: refused},
		{name: "double slash", rules: publicThenAdmin, target: "/public//../admin/users", assert: refused},
		{name: "encoded slash, default config", rules: publicThenAdmin, target: "/public%2F..%2Fadmin/users", assert: refused},
		{
			name: "encoded slash under UnescapePath", config: fiber.Config{UnescapePath: true},
			rules: publicThenAdmin, target: "/public%2F..%2Fadmin/users", assert: refused,
		},
		{name: "semicolon dot-dot", rules: publicThenAdmin, target: "/public/..;/admin/users", assert: refused},
		{name: "backslash dot-dot", rules: publicThenAdmin, target: `/public/..\admin/users`, assert: refused},
	}

	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			t.Parallel()

			az := NewMockAuthorizer(gomock.NewController(t))
			chain := newChain(t,
				httpsec.RegisterInterceptor(publishing(testPrincipal()), httpsec.OrderBearerToken),
				httpsec.EnableAuthorization(az, tc.rules...),
			)

			var reached atomic.Bool

			cfg := tc.config
			cfg.ErrorHandler = fibersec.ErrorHandler
			app := fiber.New(cfg)
			app.Use(fibersec.Middleware(chain))
			app.Get("/admin/users", func(c fiber.Ctx) error {
				reached.Store(true)
				return c.SendString("admin")
			})

			tc.assert(t, sendRawTarget(t, app, &reached, http.MethodGet, tc.target, ""))
		})
	}
}

// TestFiber_RecoveryCooldownSeesRoutedPath pins the cool-down's promise that
// no framework's lenient routing reaches a marked route by a path the guard
// does not match.
func TestFiber_RecoveryCooldownSeesRoutedPath(t *testing.T) {
	t.Parallel()

	type testCase struct {
		name   string
		target string
		assert func(t *testing.T, r rawTargetResult)
	}

	refused := func(t *testing.T, r rawTargetResult) {
		t.Helper()
		assert.False(t, r.reached, "the marked handler ran (status %d)", r.status)
		assert.Equal(t, http.StatusForbidden, r.status)
	}

	cases := []testCase{
		{name: "control: canonical path", target: "/account/delete", assert: refused},
		{name: "control: upper case", target: "/ACCOUNT/Delete/", assert: refused},
		{name: "dot-dot segment", target: "/x/../account/delete", assert: refused},
		{name: "dot segment", target: "/account/./delete", assert: refused},
		{name: "encoded dot segment", target: "/account/%2e/delete", assert: refused},
	}

	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			t.Parallel()

			rid, err := id.NewV7Generator().NewID()
			require.NoError(t, err)

			at := time.Now().Add(-time.Minute)
			records := recovery.NewMemoryRecordStore()
			require.NoError(t, records.Insert(t.Context(), recovery.Record{
				ID: rid, User: identity.UserID("u-1"), StartedAt: at, NotBefore: at, CompletedAt: at,
			}))

			opts, _ := authenticated(t)
			opts = append(opts, httpsec.EnableRecoveryCooldown(records, time.Hour,
				httpsec.Route{Method: http.MethodPost, Path: "/account/delete"}))
			chain := newChain(t, opts...)

			var reached atomic.Bool

			app := fiber.New(fiber.Config{ErrorHandler: fibersec.ErrorHandler})
			app.Use(fibersec.Middleware(chain))
			app.Post("/account/delete", func(c fiber.Ctx) error {
				reached.Store(true)
				return c.SendString("deleted")
			})

			tc.assert(t, sendRawTarget(t, app, &reached, http.MethodPost, tc.target,
				"Authorization: Bearer "+presentedToken+"\r\n"))
		})
	}
}
